def calculate_day_trade(buy_price, sell_price, discount=0.25, shares=1000): # 基本參數 tax_rate = 0.0015 # 當沖交易稅減半為 0.15% fee_rate = 0.001425 # 標準手續費率 # 1. 買進成本 buy_raw_fee = buy_price * shares * fee_rate buy_fee = int(buy_raw_fee * discount) if buy_fee < 20: buy_fee = 20 # 低消通常為 20 元 total_cost = (buy_price * shares) + buy_fee # 2. 賣出收入 sell_raw_fee = sell_price * shares * fee_rate sell_fee = int(sell_raw_fee * discount) if sell_fee < 20: sell_fee = 20 sell_tax = int(sell_price * shares * tax_rate) total_revenue = (sell_price * shares) - sell_fee - sell_tax # 3. 損益計算 profit = total_revenue - total_cost return profit, buy_fee + sell_fee + sell_tax def get_tick_size(price): """計算台股升降單位 (Tick)""" if price < 10: return 0.01 elif price < 50: return 0.05 elif price < 100: return 0.1 elif price < 500: return 0.5 elif price < 1000: return 1.0 else: return 5.0 # --- 執行試算 --- buy_p = float(input("請輸入買入價格: ")) sell_p = float(input("請輸入賣出價格: ")) profit, total_taxes = calculate_day_trade(buy_p, sell_p) tick = get_tick_size(buy_p) print("-" * 30) print(f"買入價格: {buy_p} | 賣出價格: {sell_p}") print(f"預估損益: {profit:,.0f} 元") print(f"總交易成本 (手續費+稅): {total_taxes:,.0f} 元") # 標示幾個 Tick 獲利 print("-" * 30) print("【跳動檔位試算】") for i in range(1, 6): target_p = buy_p + (tick * i) p, _ = calculate_day_trade(buy_p, target_p) status = "盈" if p > 0 else "虧" print(f"跳 {i} 檔 ({target_p:.2f}): 損益約 {p:,.0f} 元 ({status})")
